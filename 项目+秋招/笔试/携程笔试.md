import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int T = scanner.nextInt();
        while (T-- > 0) {
            int n = scanner.nextInt();
            int[] a = new int[n];
            for (int i = 0; i < n; i++) {
                a[i] = scanner.nextInt();
            }

            int[] count = new int[32];
            for (int num : a) {
                for (int k = 0; k < 32; k++) {
                    if ((num >> k & 1) == 1) {
                        count[k]++;
                    }
                }
            }

            long maxSum = 0;
            for (int s : a) {
                long currentSum = 0;
                for (int k = 0; k < 32; k++) {
                    if ((s >> k & 1) == 1) {
                        currentSum += (long) (n - count[k]) * (1L << k);
                    } else {
                        currentSum += (long) count[k] * (1L << k);
                    }
                }
                if (currentSum > maxSum) {
                    maxSum = currentSum;
                }
            }

            long sumZero = 0;
            for (int k = 0; k < 32; k++) {
                sumZero += (long) count[k] * (1L << k);
            }
            if (sumZero > maxSum) {
                maxSum = sumZero;
            }

            System.out.println(maxSum);
        }
    }
}

import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        
        long totalInv = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (a[i] > a[j]) {
                    totalInv++;
                }
            }
        }
        
        int[][] PreSum = new int[n][n];
        for (int r = 0; r < n; r++) {
            int[] signs = new int[n];
            for (int i = 0; i < n; i++) {
                if (a[i] < a[r]) {
                    signs[i] = 1;
                } else if (a[i] > a[r]) {
                    signs[i] = -1;
                } else {
                    signs[i] = 0;
                }
            }
            PreSum[r][0] = signs[0];
            for (int i = 1; i < n; i++) {
                PreSum[r][i] = PreSum[r][i-1] + signs[i];
            }
        }
        
        long minDiff = 0;
        for (int l = 0; l < n; l++) {
            long current = 0;
            for (int r = l; r < n; r++) {
                if (r > l) {
                    int s = PreSum[r][r-1];
                    if (l > 0) {
                        s -= PreSum[r][l-1];
                    }
                    current += s;
                }
                if (current < minDiff) {
                    minDiff = current;
                }
            }
        }
        
        System.out.println(totalInv + minDiff);
    }
}
