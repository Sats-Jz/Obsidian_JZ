### 序列函数
1. NTILE:ntile是Hive很强大的一个分析函数。可以看成是：它把有序的数据集合平均分配到指定的数量(num)个桶，俑号分配给每一行。如果不能平均分配，则优先分配较小编号的桶，并且各个桶中能放的行数最多相差1

```sql
# 获取每个人消费记录的后50%的消费记录
with t as(select name, orderdate,cost, ntile(2) over(partirion by name) from
          )
select * from t where xh = 2
```

```java
// 实现大写变小写
public class ToUpperUDF extends UDF { // 基于反射
    
    public Text evaluate(Text input) {
        if (input == null) return null;
        return new Text(input.toString().toUpperCase());
    }
    
    // 可以重载多个evaluate方法处理不同类型
    public String evaluate(String input) {
        if (input == null) return null;
        return input.toUpperCase();
    }
    
    // 处理多个参数
    public Text evaluate(Text input, Text suffix) {
        if (input == null) return null;
        String result = input.toString().toUpperCase();
        if (suffix != null) {
            result += suffix.toString();
        }
        return new Text(result);
    }
}
```

```java
public class LowerString extends GenericUDF {// 性能更好
    
    private StringObjectInspector stringInspector;
    // 校验参数
    @Override
    public ObjectInspector initialize(ObjectInspector[] arguments) throws UDFArgumentException {
        // 1. 检查参数个数
        if (arguments.length != 1) {
            throw new UDFArgumentLengthException("LowerString函数只需要1个参数");
        }
        
        // 2. 检查参数类型
        if (arguments[0].getCategory() != Category.PRIMITIVE) {
            throw new UDFArgumentException("LowerString函数只接受基本类型参数");
        }
        
        // 3. 获取字符串检查器
        stringInspector = (StringObjectInspector) arguments[0];
        
        // 4. 返回输出类型（字符串）
        return PrimitiveObjectInspectorFactory.writableStringObjectInspector;
    }
    // 核心实现
    @Override
    public Object evaluate(DeferredObject[] arguments) throws HiveException {
        if (arguments[0].get() == null) {
            return null;
        }
        
        // 获取输入字符串
        String input = stringInspector.getPrimitiveJavaObject(arguments[0].get());
        
        // 转换为小写并返回
        return new Text(input.toLowerCase());
    }
    // 函数干什么的

    @Override
    public String getDisplayString(String[] children) {
        return "大写字母变小写字母";
    }
}
```
